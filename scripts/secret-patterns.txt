# Cloud / provider access keys

# GitHub tokens
gh[pousr]_[A-Za-z0-9]{36,}
github_pat_[A-Za-z0-9_]{22,}

# Slack tokens
xox[baprs]-[A-Za-z0-9-]{10,}

# OpenAI/Mistral-style API keys (sk-... with a long random-looking suffix;

sk-(proj-)?[A-Za-z0-9_-]{20,}

# JWTs
eyJ[A-Za-z0-9_-]{10,}\.[A-Za-z0-9_-]{10,}\.[A-Za-z0-9_-]{5,}

# Generic credential assignments: name, then : or =, then a quoted value of

# templates do not match; obvious test fixtures are filtered by
# secret-scan-exclusions.txt)
(password|passwd|secret|api[_-]?key|apikey|access[_-]?key|secret[_-]?key|client[_-]?secret|auth[_-]?token|access[_-]?token|refresh[_-]?token|private[_-]?token)[[:space:]]*[:=][[:space:]]*["'][^"']{8,}["']
